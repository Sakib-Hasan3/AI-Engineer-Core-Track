# 🧠 5. Define Tools for OpenAI (Function Calling)

```python
tools = [
    {
        "type": "function",
        "function": {
            "name": "search_flights",
            "description": "Search available flights",
            "parameters": {
                "type": "object",
                "properties": {
                    "source": {"type": "string"},
                    "destination": {"type": "string"},
                    "date": {"type": "string"}
                },
                "required": ["source", "destination", "date"]
            }
        }
    },
    {
        "type": "function",
        "function": {
            "name": "flight_status",
            "description": "Check flight status",
            "parameters": {
                "type": "object",
                "properties": {
                    "flight_no": {"type": "string"}
                },
                "required": ["flight_no"]
            }
        }
    }
]
```

# 🤖 6. OpenAI Tool Calling Logic

```python
from openai import OpenAI
import json

client = OpenAI()

def agent_response(user_input):
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": user_input}],
        tools=tools,
        tool_choice="auto"
    )

    message = response.choices[0].message

    # If tool is called
    if message.tool_calls:
        tool_call = message.tool_calls[0]
        function_name = tool_call.function.name
        args = json.loads(tool_call.function.arguments)

        # Execute tool
        if function_name == "search_flights":
            result = search_flights(**args)

        elif function_name == "flight_status":
            result = flight_status(**args)

        # Send result back to LLM
        final = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "user", "content": user_input},
                message,
                {
                    "role": "tool",
                    "tool_call_id": tool_call.id,
                    "content": str(result)
                }
            ]
        )

        return final.choices[0].message.content

    return message.content
```

# 🎛️ 7. Gradio UI

```python
import gradio as gr

def chat(user_input):
    return agent_response(user_input)

ui = gr.Interface(
    fn=chat,
    inputs="text",
    outputs="text",
    title="✈️ Airline AI Assistant",
    description="Search flights, check status, and book tickets using AI tools"
)

ui.launch()
```
